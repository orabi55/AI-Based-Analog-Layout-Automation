% --- Ornamental Rule (Enhanced with diamond pattern) ---
\newcommand{\ornamentalrule}{
  \begin{center}
    \begin{tikzpicture}
      \draw[very thick, WarmGold] (0,0) -- (2.2,0);
      \draw[very thick, WarmGold] (3.8,0) -- (6,0);
      % Central diamond ornament
      \filldraw[WarmGold] (3,0) -- (3,-0.12) -- (2.8,0) -- (3,0.12) -- cycle;
      \filldraw[CopperAccent] (3,0) circle (1.5pt);
      % Side diamonds
      \filldraw[WarmGold] (2.4,0) -- (2.4,-0.06) -- (2.3,0) -- (2.4,0.06) -- cycle;
      \filldraw[WarmGold] (3.6,0) -- (3.6,-0.06) -- (3.7,0) -- (3.6,0.06) -- cycle;
      % Thin parallel accent lines
      \draw[thin, WarmGold!60] (0.3,0.08) -- (2.1,0.08);
      \draw[thin, WarmGold!60] (3.9,0.08) -- (5.7,0.08);
    \end{tikzpicture}
  \end{center}
}

% --- Grand Ornamental Separator (for title pages) ---
\newcommand{\grandornament}{
  \begin{center}
    \begin{tikzpicture}
      % Main horizontal lines
      \draw[ultra thick, WarmGold] (-3.5,0) -- (-0.8,0);
      \draw[ultra thick, WarmGold] (0.8,0) -- (3.5,0);
      % Central ornament - layered diamonds
      \filldraw[WarmGold] (0,0) -- (0.25,-0.2) -- (0,-0.4) -- (-0.25,-0.2) -- cycle;
      \filldraw[AcademicBlue] (0,-0.05) -- (0.18,-0.2) -- (0,-0.35) -- (-0.18,-0.2) -- cycle;
      \filldraw[WarmGold] (0,-0.15) circle (2pt);
      % Side ornaments
      \foreach \x in {-0.6, 0.6} {
        \filldraw[WarmGold] (\x,0) circle (2pt);
      }
      % Decorative curls at ends
      \draw[thick, WarmGold, decorate, decoration={snake, amplitude=1pt, segment length=6pt}] (-3.5,0) -- (-3.8,0);
      \draw[thick, WarmGold, decorate, decoration={snake, amplitude=1pt, segment length=6pt}] (3.5,0) -- (3.8,0);
    \end{tikzpicture}
  \end{center}
}

% --- Corner Filigree (for decorative page frames) ---
% Usage: \cornerfiligree{position}{rotation}
% position: e.g., (current page.north west), rotation: angle
\newcommand{\cornerfiligree}[2]{
  \begin{scope}[shift={#1}, rotate=#2]
    \draw[thick, WarmGold, opacity=0.6] (0,0) -- (1.5,0);
    \draw[thick, WarmGold, opacity=0.6] (0,0) -- (0,-1.5);
    \draw[thick, WarmGold, opacity=0.4] (0.15,-0.15) -- (1.2,-0.15);
    \draw[thick, WarmGold, opacity=0.4] (0.15,-0.15) -- (0.15,-1.2);
    % Corner flourish
    \draw[thick, WarmGold, opacity=0.5] (0.3,0) arc (0:-90:0.3);
    \filldraw[WarmGold, opacity=0.7] (0,0) circle (2.5pt);
  \end{scope}
}

% --- Decorative Page Frame ---
\newcommand{\decorativeframe}{
  \begin{tikzpicture}[remember picture, overlay]
    % Outer frame
    \draw[line width=1.5pt, WarmGold]
      ($(current page.north west) + (1.8cm,-1.5cm)$)
      rectangle
      ($(current page.south east) + (-1.5cm,1.5cm)$);
    % Inner frame (double-line effect)
    \draw[line width=0.5pt, WarmGold!50]
      ($(current page.north west) + (1.95cm,-1.65cm)$)
      rectangle
      ($(current page.south east) + (-1.65cm,1.65cm)$);
    % Corner filigrees
    \cornerfiligree{($(current page.north west) + (1.8cm,-1.5cm)$)}{0}
    \cornerfiligree{($(current page.north east) + (-1.5cm,-1.5cm)$)}{-90}
    \cornerfiligree{($(current page.south east) + (-1.5cm,1.5cm)$)}{180}
    \cornerfiligree{($(current page.south west) + (1.8cm,1.5cm)$)}{90}
  \end{tikzpicture}
}
